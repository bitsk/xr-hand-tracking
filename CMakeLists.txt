cmake_minimum_required(VERSION 3.16)
project(xr_hand_tracking VERSION 0.1.1 LANGUAGES NONE)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(xr_hand_api INTERFACE)
set_target_properties(xr_hand_api PROPERTIES EXPORT_NAME api)
target_include_directories(xr_hand_api INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS xr_hand_api EXPORT XRHandTargets)
install(EXPORT XRHandTargets NAMESPACE XRHand:: DESTINATION ${CMAKE_INSTALL_DATADIR}/cmake/XRHand)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/XRHandConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/XRHandConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_DATADIR}/cmake/XRHand)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/XRHandConfig.cmake DESTINATION ${CMAKE_INSTALL_DATADIR}/cmake/XRHand)
install(DIRECTORY src/xr_hand DESTINATION ${CMAKE_INSTALL_DATADIR}/xr-hand-tracking/python PATTERN __pycache__ EXCLUDE)
install(PROGRAMS tools/xr-hand-trackingd tools/xr-hand-status tools/xr-hand-import-assets DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES systemd/xr-hand-tracking.service DESTINATION lib/systemd/system)
install(FILES config/hand-tracking.json DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/xr)
option(XR_HAND_BUILD_QUEST_BACKEND "Build optional separately packaged Quest backend" OFF)
if(XR_HAND_BUILD_QUEST_BACKEND)
  add_subdirectory(backends/quest)
endif()
